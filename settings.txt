local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

local function waitForGUI()
	local start = os.clock()
	while os.clock() - start < 12 do
		local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui")
		if pg then
			local g = pg:FindFirstChild("AutoPlayerStatus")
			if g then return g end
		end
		task.wait(0.15)
	end
end

local rows = {}

local function updateTotal()
	local sum = 0
	for _, r in ipairs(rows) do
		sum = sum + (tonumber(r.pct.Text) or 0)
	end
	total.Text = "Total: " .. sum .. "%"
	total.TextColor3 = sum == 100 and Color3.fromRGB(140, 255, 160) or Color3.fromRGB(255, 120, 120)
	return sum
end

local function addRow(data)
	data = data or {min = 0, max = 30, percent = 50}

	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, -10, 0, 36)
	row.BackgroundColor3 = Color3.fromRGB(28, 28, 36)
	row.Parent = list

	local minBox = Instance.new("TextBox")
	minBox.Size = UDim2.new(0, 42, 0, 22)
	minBox.Position = UDim2.new(0, 8, 0, 7)
	minBox.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
	minBox.Text = tostring(data.min)
	minBox.TextColor3 = Color3.fromRGB(230, 230, 240)
	minBox.Font = Enum.Font.Code
	minBox.TextSize = 12
	minBox.Parent = row

	local maxBox = Instance.new("TextBox")
	maxBox.Size = UDim2.new(0, 42, 0, 22)
	maxBox.Position = UDim2.new(0, 56, 0, 7)
	maxBox.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
	maxBox.Text = tostring(data.max)
	maxBox.TextColor3 = Color3.fromRGB(230, 230, 240)
	maxBox.Font = Enum.Font.Code
	maxBox.TextSize = 12
	maxBox.Parent = row

	local pctBox = Instance.new("TextBox")
	pctBox.Size = UDim2.new(0, 42, 0, 22)
	pctBox.Position = UDim2.new(0, 108, 0, 7)
	pctBox.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
	pctBox.Text = tostring(data.percent)
	pctBox.TextColor3 = Color3.fromRGB(230, 230, 240)
	pctBox.Font = Enum.Font.Code
	pctBox.TextSize = 12
	pctBox.Parent = row

	local rem = Instance.new("TextButton")
	rem.Size = UDim2.new(0, 22, 0, 22)
	rem.Position = UDim2.new(1, -30, 0, 7)
	rem.BackgroundColor3 = Color3.fromRGB(60, 25, 35)
	rem.Text = "×"
	rem.TextColor3 = Color3.fromRGB(255, 140, 140)
	rem.Font = Enum.Font.GothamBold
	rem.TextSize = 14
	rem.Parent = row

	local dataRow = {min = minBox, max = maxBox, pct = pctBox, frame = row}
	table.insert(rows, dataRow)

	rem.MouseButton1Click:Connect(function()
		row:Destroy()
		for i = #rows, 1, -1 do
			if rows[i] == dataRow then
				table.remove(rows, i)
				break
			end
		end
		list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 8)
		updateTotal()
	end)

	minBox:GetPropertyChangedSignal("Text"):Connect(updateTotal)
	maxBox:GetPropertyChangedSignal("Text"):Connect(updateTotal)
	pctBox:GetPropertyChangedSignal("Text"):Connect(updateTotal)

	list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 8)
	updateTotal()
end

local cur = getgenv().AutoPlayerDelayRanges or {
	{min = 0, max = 20, percent = 90},
	{min = 21, max = 50, percent = 10}
}

for _, v in ipairs(cur) do
	addRow(v)
end

add.MouseButton1Click:Connect(function()
	addRow()
end)

apply.MouseButton1Click:Connect(function()
	if updateTotal() \~= 100 then
		total.Text = "MUST BE 100% !"
		total.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local new = {}
	for _, r in ipairs(rows) do
		local mn = tonumber(r.min.Text) or 0
		local mx = tonumber(r.max.Text) or mn
		local pc = tonumber(r.pct.Text) or 0
		if mx < mn then mx = mn end
		table.insert(new, {
			min = math.floor(mn),
			max = math.floor(mx),
			percent = pc
		})
	end

	getgenv().AutoPlayerDelayRanges = new
	total.Text = "Applied ✓"
	total.TextColor3 = Color3.fromRGB(140, 255, 160)
end)

close.MouseButton1Click:Connect(function()
	panel.Visible = false
end)

btn.MouseButton1Click:Connect(function()
	panel.Visible = not panel.Visible
end)
